<?php

namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Symfony\Component\HttpFoundation\Response;

class IsAdmin
{
    public function handle(Request $request, Closure $next): Response
    {
        if (Auth::check() && Auth::user()->role === 'admin') {
            return $next($request);
        }

        return redirect('/')->with('error', 'Akses ditolak! Halaman ini khusus Administrator.');
    }
}

📄 File: app/Http/Controllers/TicketController.php [TAMBAHKAN / UBAH] 
<?php

namespace App\Http\Controllers;

use App\Models\Ticket;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Storage;

class TicketController extends Controller
{
    // 1. Tampilkan tabel master tiket konser
    public function index()
    {
        $tickets = Ticket::latest()->paginate(10);
        return view('admin.tickets.index', compact('tickets'));
    }

    // 2. Tampilkan form tambah tiket
    public function create()
    {
        return view('admin.tickets.create');
    }

    // 3. Simpan tiket baru + upload poster fisik
    public function store(Request $request)
    {
        $request->validate([
            'concert_name' => 'required|string|max:255',
            'artist'       => 'required|string|max:255',
            'venue'        => 'required|string|max:255',
            'event_date'   => 'required|date',
            'category'     => 'required|in:VIP,Festival,Regular',
            'price'        => 'required|numeric|min:0',
            'stock'        => 'required|integer|min:1',
            'description'  => 'nullable|string',
            'poster'       => 'nullable|image|mimes:jpeg,png,jpg|max:2048',
        ]);

        $posterPath = null;
        if ($request->hasFile('poster')) {
            $posterPath = $request->file('poster')->store('posters', 'public');
        }

        Ticket::create([
            'concert_name' => $request->concert_name,
            'artist'       => $request->artist,
            'venue'        => $request->venue,
            'event_date'   => $request->event_date,
            'category'     => $request->category,
            'price'        => $request->price,
            'stock'        => $request->stock,
            'description'  => $request->description,
            'poster'       => $posterPath,
        ]);

        return redirect()->route('admin.tickets.index')->with('success', 'Tiket konser berhasil ditambahkan!');
    }

    // 4. Tampilkan form edit tiket
    public function edit(Ticket $ticket)
    {
        return view('admin.tickets.edit', compact('ticket'));
    }

    // 5. Update data tiket + ganti file poster lama
    public function update(Request $request, Ticket $ticket)
    {
        $request->validate([
            'concert_name' => 'required|string|max:255',
            'artist'       => 'required|string|max:255',
            'venue'        => 'required|string|max:255',
            'event_date'   => 'required|date',
            'category'     => 'required|in:VIP,Festival,Regular',
            'price'        => 'required|numeric|min:0',
            'stock'        => 'required|integer|min:0',
            'description'  => 'nullable|string',
            'poster'       => 'nullable|image|mimes:jpeg,png,jpg|max:2048',
        ]);

        $posterPath = $ticket->poster;
        if ($request->hasFile('poster')) {
            if ($ticket->poster && Storage::disk('public')->exists($ticket->poster)) {
                Storage::disk('public')->delete($ticket->poster);
            }
            $posterPath = $request->file('poster')->store('posters', 'public');
        }

        $ticket->update([
            'concert_name' => $request->concert_name,
            'artist'       => $request->artist,
            'venue'        => $request->venue,
            'event_date'   => $request->event_date,
            'category'     => $request->category,
            'price'        => $request->price,
            'stock'        => $request->stock,
            'description'  => $request->description,
            'poster'       => $posterPath,
        ]);

        return redirect()->route('admin.tickets.index')->with('success', 'Data tiket konser berhasil diperbarui!');
    }

    // 6. Hapus tiket + hapus poster fisik dari disk
    public function destroy(Ticket $ticket)
    {
        if ($ticket->poster && Storage::disk('public')->exists($ticket->poster)) {
            Storage::disk('public')->delete($ticket->poster);
        }
        $ticket->delete();

        return redirect()->route('admin.tickets.index')->with('success', 'Tiket konser berhasil dihapus!');
    }
}

[LANGKAH 2.5] Controller CRUD User Management (Admin & Customer)

📄 File: app/Http/Controllers/Admin/UserController.php [TAMBAHKAN / UBAH] 
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;

class UserController extends Controller
{
    public function index()
    {
        $users = User::latest()->paginate(10);
        return view('admin.users.index', compact('users'));
    }

    public function create()
    {
        return view('admin.users.create');
    }

    public function store(Request $request)
    {
        $request->validate([
            'name'     => 'required|string|max:255',
            'email'    => 'required|email|unique:users',
            'phone'    => 'nullable|string|max:20',
            'password' => 'required|min:6',
            'role'     => 'required|in:admin,customer',
        ]);

        User::create([
            'name'     => $request->name,
            'email'    => $request->email,
            'phone'    => $request->phone,
            'password' => Hash::make($request->password),
            'role'     => $request->role,
        ]);

        return redirect()->route('admin.users.index')->with('success', 'User berhasil ditambahkan!');
    }

    public function edit(User $user)
    {
        return view('admin.users.edit', compact('user'));
    }

    public function update(Request $request, User $user)
    {
        $request->validate([
            'name'  => 'required|string|max:255',
            'email' => 'required|email|unique:users,email,' . $user->id,
            'phone' => 'nullable|string|max:20',
            'role'  => 'required|in:admin,customer',
        ]);

        $data = [
            'name'  => $request->name,
            'email' => $request->email,
            'phone' => $request->phone,
            'role'  => $request->role,
        ];

        if ($request->filled('password')) {
            $request->validate(['password' => 'min:6']);
            $data['password'] = Hash::make($request->password);
        }

        $user->update($data);
        return redirect()->route('admin.users.index')->with('success', 'Data user berhasil diperbarui!');
    }

    public function destroy(User $user)
    {
        // Proteksi Anti Self-Delete
        if (auth()->id() === $user->id) {
            return back()->with('error', 'Anda tidak dapat menghapus akun Anda sendiri!');
        }

        $user->delete();
        return redirect()->route('admin.users.index')->with('success', 'User berhasil dihapus!');
    }
}

[LANGKAH 3.1] Controller Transaksi Tiket Konser (DB::transaction)

Buka app/Http/Controllers/BookingController.php. Checkout menggunakan DB::transaction dan otomatis mengurangi kuota tiket (decrement):

📄 File: app/Http/Controllers/BookingController.php [TAMBAHKAN / UBAH] 
<?php

namespace App\Http\Controllers;

use App\Models\Ticket;
use App\Models\Order;
use App\Models\OrderDetail;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Auth;

class BookingController extends Controller
{
    // 1. Katalog Tiket Konser Publik
    public function index(Request $request)
    {
        $query = Ticket::where('stock', '>', 0);

        // Filter Kategori
        if ($request->filled('category') && $request->category !== 'all') {
            $query->where('category', $request->category);
        }

        $tickets = $query->latest()->get();
        return view('customer.concerts', compact('tickets'));
    }

    // 2. Checkout Pemesanan Tiket (DB::transaction & Auto Decrement Stok)
    public function store(Request $request)
    {
        $request->validate([
            'ticket_id'      => 'required|exists:tickets,id',
            'quantity'       => 'required|integer|min:1',
            'customer_name'  => 'required|string|max:255',
            'customer_phone' => 'required|string|max:20',
        ]);

        DB::beginTransaction();
        try {
            $ticket = Ticket::findOrFail($request->ticket_id);

            // Cek Ketersediaan Stok Kuota Tiket
            if ($ticket->stock < $request->quantity) {
                return back()->with('error', "Gagal! Sisa kuota tiket hanya tersisa {$ticket->stock} kursi.");
            }

            $subtotal = $ticket->price * $request->quantity;
            $orderCode = 'TKT-' . date('Ymd') . '-' . strtoupper(substr(uniqid(), -4));

            // Simpan Header Order
            $order = Order::create([
                'user_id'        => Auth::id() ?? null,
                'order_code'     => $orderCode,
                'customer_name'  => $request->customer_name,
                'customer_phone' => $request->customer_phone,
                'total_price'    => $subtotal,
                'status'         => 'paid',
                'order_date'     => now(),
            ]);

            // Simpan Detail Order
            OrderDetail::create([
                'order_id'   => $order->id,
                'ticket_id'  => $ticket->id,
                'quantity'   => $request->quantity,
                'subtotal'   => $subtotal,
            ]);

            // Kurangi Stok Kuota Tiket Konser
            $ticket->decrement('stock', $request->quantity);

            DB::commit();

            return redirect()->route('customer.concerts')
                ->with('success', "Pesanan tiket {$ticket->concert_name} berhasil! Kode Booking: {$order->order_code}");

        } catch (\Exception $e) {
            DB::rollBack();
            return back()->with('error', 'Gagal memproses pesanan: ' . $e->getMessage());
        }
    }
}

<?php

namespace App\Http\Controllers;

use App\Models\Order;
use Illuminate\Http\Request;

class OrderReportController extends Controller
{
    // 1. Dashboard Rekap Transaksi Tiket Masuk
    public function index(Request $request)
    {
        $query = Order::with('orderDetails.ticket')->latest();

        // Filter Periode Tanggal
        if ($request->filled('start_date') && $request->filled('end_date')) {
            $query->whereBetween('order_date', [$request->start_date, $request->end_date]);
        }

        // Filter Status Bayar
        if ($request->filled('status')) {
            $query->where('status', $request->status);
        }

        $orders = $query->paginate(15);

        // Perhitungan Statistik Otomatis
        $totalIncome = Order::where('status', 'paid')->sum('total_price');
        $totalOrders = Order::count();
        $pendingOrders = Order::where('status', 'pending')->count();

        return view('dashboard', compact('orders', 'totalIncome', 'totalOrders', 'pendingOrders'));
    }

    // 2. Ubah Status Transaksi Pesanan
    public function updateStatus(Request $request, $id)
    {
        $request->validate(['status' => 'required|in:pending,paid,cancelled']);
        $order = Order::findOrFail($id);
        $order->update(['status' => $request->status]);

        return back()->with('success', "Status pesanan {$order->order_code} berhasil diubah!");
    }
}

MODUL 5: KISI-KISI TEORI & WAWANCARA ASESOR BNSP

Jawaban lugas dan berbobot untuk 4 pertanyaan teori yang paling sering diajukan asesor:

1. Apa Saja yang Dibutuhkan dalam Membangun Aplikasi Pemesanan Tiket Konser?

Jelaskan secara terstruktur ke asesor terbagi menjadi 4 komponen:

  1. Kebutuhan Hardware: PC/Laptop minimal Core i3/Ryzen 3, RAM 8 GB (agar lancar menjalankan XAMPP, VS Code, dan browser), dan SSD min. 10 GB.
  2. Kebutuhan Software & Environment:
    • Web Server & Database: XAMPP (Apache & MySQL / MariaDB).
    • Bahasa Pemrograman: PHP 8.x dan framework Laravel 10/11.
    • Package Manager: Composer (PHP) dan Node.js / NPM (Frontend Vite).
    • Code Editor & Browser: VS Code dan Chrome DevTools.
  3. Kebutuhan Database: RDBMS MySQL dengan perancangan 4 tabel relasional (users, tickets, orders, order_details) dengan mesin penyimpanan InnoDB yang mendukung relasi Foreign Key cascade dan transaksi ACID.
  4. Kebutuhan Fungsional & Keamanan: Autentikasi multi-role (Admin vs Customer), manajemen kuota tiket konser real-time, transaksi atomik (DB::transaction), dan proteksi keamanan (CSRF Token, Hash Password Bcrypt).
2. Apa Fungsi ERD dan Fungsi Tambah, Edit, Hapus, Tampil dalam Bahasa SQL Beserta Contohnya?

A. Fungsi ERD (Entity Relationship Diagram):

  • Blueprint Perancangan: Sebagai cetak biru visual struktur tabel sebelum dikodekan ke dalam migrasi Laravel.
  • Menentukan Kardinalitas Relasi: Menentukan hubungan 1:N (1 User punya banyak Order, 1 Order punya banyak OrderDetail).
  • Mencegah Redundansi Data: Dasar proses normalisasi basis data sehingga tidak ada duplikasi data transaksi.

B. Fungsi CRUD dalam Bahasa SQL (Kasus Tiket Konser):

  • Create (Tambah - INSERT INTO): Menambahkan baris data tiket baru.
    INSERT INTO tickets (concert_name, artist, venue, event_date, category, price, stock, created_at, updated_at) VALUES ('Coldplay Jakarta', 'Coldplay', 'GBK', '2026-11-15', 'VIP', 3500000, 50, NOW(), NOW());
  • Read (Tampil - SELECT): Menampilkan dan menyaring data tiket.
    SELECT id, concert_name, category, price, stock FROM tickets WHERE category = 'VIP' AND stock > 0 ORDER BY price ASC;
  • Update (Edit - UPDATE ... SET ... WHERE): Mengubah data kuota atau status pesanan.
    UPDATE tickets SET stock = stock - 2, updated_at = NOW() WHERE id = 1;
    *Catatan: Wajib pakai WHERE agar tidak mengubah seluruh baris tabel!
  • Delete (Hapus - DELETE FROM ... WHERE): Menghapus data tiket yang batal.
    DELETE FROM tickets WHERE id = 1;
    *Catatan: Wajib pakai WHERE agar tidak menghapus seluruh isi tabel!
3. Apa yang Dimaksud Query, Fungsinya, Beserta Contohnya?

Pengertian Query: Permintaan instruksi terstruktur (menggunakan sintaks SQL) yang dikirimkan ke DBMS untuk meminta, memfilter, mengolah, memanipulasi, atau merekap data dari tabel database.

Fungsi & Manfaat Utama:

  1. Penyaringan Data Spesifik: Menampilkan data yang tepat menggunakan klausul WHERE atau BETWEEN tanpa membebani memori.
  2. Penggabungan Multi-Tabel (JOIN): Mengintegrasikan data dari tabel terpisah (misal: menggabungkan nomor order dengan nama konser dari tabel tickets).
  3. Kalkulasi Agregasi Cepat: Menghitung omset pendapatan secara instan dengan fungsi SUM, COUNT, AVG di level database.

Contoh Query Kompleks (Laporan Omset Konser):

SELECT t.concert_name, t.category, SUM(od.quantity) AS tiket_terjual, SUM(od.subtotal) AS total_omset FROM order_details od INNER JOIN tickets t ON od.ticket_id = t.id INNER JOIN orders o ON od.order_id = o.id WHERE o.status = 'paid' GROUP BY t.id, t.concert_name, t.category;
4. Jelaskan Konsep dan Arsitektur MVC pada Framework Laravel!

MVC (Model - View - Controller) adalah pola arsitektur perangkat lunak yang memisahkan logika aplikasi menjadi 3 komponen utama:

  • Model (M): Mengelola logika data, relasi basis data via Eloquent ORM, dan aturan penyimpanan ke tabel MySQL (contoh: Ticket.php, Order.php).
  • View (V): Mengatur presentasi antarmuka pengguna (UI) menggunakan Blade template engine untuk menampilkan data ke layar browser (contoh: concerts.blade.php).
  • Controller (C): Jembatan penerima HTTP Request, pengatur alur validasi input, pemrosesan transaksi data ke Model, lalu mengirimkan hasilnya ke View (contoh: TicketController.php).

Alur Lifecycle Request Laravel:

Browser Request → routes/web.php → Middleware → Controller → Model (MySQL) → View (Blade) → Response HTML Browser.

MassAssignmentException Kolom model diblokir mass assignment. Tambahkan di Model: protected $guarded = ['id'];.
